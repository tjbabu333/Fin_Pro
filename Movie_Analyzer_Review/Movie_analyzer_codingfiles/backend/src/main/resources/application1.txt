from pydantic_settings import BaseSettings, SettingsConfigDict
from pydantic import Field


class Settings(BaseSettings):

    # ==================================================
    # Server
    # ==================================================

    SERVER_HOST: str = "0.0.0.0"
    SERVER_PORT: int = 8080
    DEBUG: bool = False

    # ==================================================
    # Application
    # ==================================================

    APP_NAME: str = "Movie Review Backend"
    APP_VERSION: str = "1.0.0"

    # ==================================================
    # Database
    # ==================================================

    DB_HOST: str = "database"
    DB_PORT: int = 5432
    DB_NAME: str = "moviereviews"
    DB_USERNAME: str = "movieuser"
    DB_PASSWORD: str = "moviepass"

    DATABASE_URL: str = (
        "postgresql+psycopg2://movieuser:moviepass@database:5432/moviereviews"
    )

    # SQLAlchemy Pool

    DB_POOL_SIZE: int = 10
    DB_MAX_OVERFLOW: int = 20
    DB_POOL_TIMEOUT: int = 5
    DB_POOL_RECYCLE: int = 1800

    # ==================================================
    # Model Server
    # ==================================================

    MODEL_SERVER_URL: str = "http://model:5000"
    MODEL_SERVER_TIMEOUT: int = 5

    # ==================================================
    # HTTP
    # ==================================================

    REQUEST_TIMEOUT: int = 30
    KEEP_ALIVE: int = 60

    # ==================================================
    # Logging
    # ==================================================

    LOG_LEVEL: str = "INFO"

    LOG_FORMAT: str = (
        "%(asctime)s | "
        "%(levelname)s | "
        "%(name)s | "
        "%(message)s"
    )

    # ==================================================
    # Security
    # ==================================================

    SECRET_KEY: str = "CHANGE_ME"

    COOKIE_HTTP_ONLY: bool = True

    COOKIE_SECURE: bool = True

    ENABLE_HSTS: bool = True

    ENABLE_CORS: bool = True

    CORS_ORIGINS: list[str] = ["*"]

    # ==================================================
    # Rate Limiting
    # ==================================================

    RATE_LIMIT = 100

    RATE_PERIOD = 60

    # ==================================================
    # Health Check
    # ==================================================

    ENABLE_HEALTH = True

    model_config = SettingsConfigDict(
        env_file=".env",
        extra="ignore",
    )


settings = Settings()

from starlette.middleware.base import BaseHTTPMiddleware


class SecurityHeadersMiddleware(BaseHTTPMiddleware):

    async def dispatch(self, request, call_next):

        response = await call_next(request)

        response.headers["X-Content-Type-Options"] = "nosniff"

        response.headers["X-Frame-Options"] = "DENY"

        response.headers["Referrer-Policy"] = (
            "strict-origin-when-cross-origin"
        )

        response.headers["Cache-Control"] = (
            "no-cache, no-store"
        )

        response.headers["Pragma"] = "no-cache"

        response.headers["Expires"] = "0"

        return response
